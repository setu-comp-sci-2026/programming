# Adding Validation to Setters

## Step 1 — The Problem Without Validation

```python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age  = age

p = Person("Alice", 25)
p.age = -5          # Python allows this — but it's invalid!
p.name = ""         # An empty name — also invalid!
print(p.age)        # -5  (bad data slipped through)
```

## Step 2 — Setter Methods

```python
class Person:
    def __init__(self, name, age):
        self._name = None
        self._age  = None
        self.set_name(name)
        self.set_age(age)

    def get_name(self):
        return self._name

    def set_name(self, name):
        if not isinstance(name, str) or not name.strip():
            raise ValueError("Name must be a non-empty string")
        self._name = name.strip().title()

    def get_age(self):
        return self._age

    def set_age(self, age):
        if not isinstance(age, int):
            raise ValueError(f"Age must be an integer, got {type(age).__name__}")
        if age < 0 or age > 150:
            raise ValueError(f"Age must be 0-150, got {age}")
        self._age = age

    def __str__(self):
        return f"Person(name='{self._name}', age={self._age})"

p = Person("  alice  ", 25)
print(p)          # Person(name='Alice', age=25)  — name was cleaned
p.set_age(26)
print(p.get_age())
```

## Step 3 — Validate in `__init__` Using Setters

In Section 01 you validated data directly inside `__init__`. That works, but if the value can be **changed later** you would have to write the same checks a second time. The cleaner pattern is to write each check **once**, in a setter, and call that setter from `__init__`:

```python
class Song:
    def __init__(self, title, artist, seconds):
        self._title   = None
        self._artist  = None
        self._seconds = None

        # Use the setters, so validation runs immediately
        self.set_title(title)
        self.set_artist(artist)
        self.set_seconds(seconds)

    def set_title(self, title):
        if not title.strip():
            raise ValueError("Title cannot be empty")
        self._title = title.strip()

    def set_artist(self, artist):
        if not artist.strip():
            raise ValueError("Artist cannot be empty")
        self._artist = artist.strip()

    def set_seconds(self, seconds):
        if not isinstance(seconds, int) or seconds <= 0:
            raise ValueError(f"Length must be a positive whole number of seconds, got {seconds}")
        self._seconds = seconds

    def get_length(self):
        """Return the length as minutes:seconds, e.g. 3:05"""
        minutes = self._seconds // 60
        secs    = self._seconds % 60
        return f"{minutes}:{secs:02d}"

    def __str__(self):
        return f"{self._title} by {self._artist} ({self.get_length()})"


song = Song("  Tramore Tides ", "The Quay Street Band", 307)
print(song)             # Tramore Tides by The Quay Street Band (5:07)

song.set_seconds(245)   # a shorter radio edit — same checks run again
print(song)             # Tramore Tides by The Quay Street Band (4:05)
```

The check for `seconds` lives in **one** place, `set_seconds()`. It runs both when the song is created and whenever the length is changed later.

**Think:** `{secs:02d}` pads the seconds with a zero. What would `get_length()` show for 245 seconds without the `02`?

## Step 4 — Testing Validation

```python
# Each of these should be rejected
invalid_cases = [
    ("", "The Quay Street Band", 200),         # empty title
    ("Tramore Tides", "   ", 200),             # blank artist
    ("Tramore Tides", "The Quay Street Band", -30),   # negative length
    ("Tramore Tides", "The Quay Street Band", 3.5),   # not a whole number
]

for title, artist, seconds in invalid_cases:
    try:
        s = Song(title, artist, seconds)
        print(f"ERROR: should have rejected ({title!r}, {artist!r}, {seconds})")
    except ValueError as e:
        print(f"Correctly rejected: {e}")

# A rejected change must leave the old value in place
song = Song("Tramore Tides", "The Quay Street Band", 307)
try:
    song.set_seconds(0)
except ValueError as e:
    print(f"Correctly rejected: {e}")
assert song.get_length() == "5:07", "A failed set_seconds() changed the length!"
print("PASS: length unchanged after a rejected change")
```

**Think:** The last test checks that a **failed** `set_seconds()` leaves the length unchanged. Which line in `set_seconds()` makes sure of that?

## Exercise 1 — Product with validation
```python
class Product:
    def __init__(self, name, price, quantity=0):
        if not name or not name.strip():
            raise ValueError("Name cannot be blank")
        if price <= 0:
            raise ValueError(f"Price must be greater than 0, got {price}")
        if quantity < 0:
            raise ValueError(f"Quantity cannot be negative, got {quantity}")
        self.name = name.strip()
        self.price = price
        self.quantity = quantity

    def restock(self, amount):
        if amount <= 0:
            raise ValueError("Restock amount must be positive")
        self.quantity += amount

    def sell(self, amount):
        if amount <= 0:
            raise ValueError("Sale amount must be positive")
        if amount > self.quantity:
            raise ValueError(f"Only {self.quantity} in stock")
        self.quantity -= amount

    def stock_value(self):
        return self.price * self.quantity

    def __str__(self):
        return (f"{self.name}: €{self.price:.2f} x {self.quantity} "
                f"(value €{self.stock_value():.2f})")


kettle = Product("Kettle", 24.99, 10)
print(kettle)          # Kettle: €24.99 x 10 (value €249.90)
kettle.sell(3)
print(kettle)          # Kettle: €24.99 x 7 (value €174.93)
kettle.restock(5)
print(kettle)          # Kettle: €24.99 x 12 (value €299.88)
try:
    kettle.sell(50)
except ValueError as e:
    print("Error:", e) # Error: Only 12 in stock
print(kettle)          # unchanged: x 12
```

## Exercise 4 — Game Character

```python
class GameCharacter:
    def __init__(self, name, health=100):
        if not name.strip():
            raise ValueError("Name cannot be empty")
        self._name = name.strip()
        self._health = 0
        self.health = health          # use the setter, so clamping happens here too

    @property
    def name(self):
        return self._name             # read-only: no setter

    @property
    def health(self):
        return self._health

    @health.setter
    def health(self, value):
        if value > 100:
            value = 100
        if value < 0:
            value = 0
        self._health = value

    @property
    def is_alive(self):
        return self._health > 0

    def take_damage(self, amount):
        if amount <= 0:
            raise ValueError("Damage must be positive")
        self.health = self._health - amount

    def heal(self, amount):
        if amount <= 0:
            raise ValueError("Healing must be positive")
        if not self.is_alive:
            raise ValueError("Cannot heal a defeated character")
        self.health = self._health + amount

    def __str__(self):
        blocks = self._health // 10
        bar = "#" * blocks + "-" * (10 - blocks)
        return f"{self._name} [{bar}] {self._health}/100"
```

The setter is the **only** place where clamping happens. `__init__`, `take_damage()` and `heal()` all assign to `self.health` rather than `self._health`, so they all go through it.

## Exercise 5 — User Account with a Password

```python
class UserAccount:
    MAX_ATTEMPTS = 3

    def __init__(self, username, password):
        if not username.strip():
            raise ValueError("Username cannot be empty")
        self._username = username.strip().lower()
        self._check_password_rules(password)
        self._password = password
        self._failed_attempts = 0

    def _check_password_rules(self, password):
        """Private helper: raise ValueError if the password breaks a rule."""
        if len(password) < 8:
            raise ValueError("Password must be at least 8 characters")
        has_digit = False
        for ch in password:
            if ch.isdigit():
                has_digit = True
        if not has_digit:
            raise ValueError("Password must contain at least one digit")

    @property
    def username(self):
        return self._username

    @property
    def is_locked(self):
        return self._failed_attempts >= UserAccount.MAX_ATTEMPTS

    def check_password(self, attempt):
        if self.is_locked:
            raise ValueError("Account is locked")
        if attempt == self._password:
            self._failed_attempts = 0
            return True
        self._failed_attempts += 1
        return False

    def change_password(self, old, new):
        if not self.check_password(old):
            raise ValueError("Old password is incorrect")
        self._check_password_rules(new)
        self._password = new

    def __str__(self):
        status = "LOCKED" if self.is_locked else "active"
        return f"User '{self._username}' ({status})"
```

Note that a wrong `old` password in `change_password()` also counts as a failed attempt, because it calls `check_password()`.

class Contact:
    def __init__(self, name, phone, email):
        self._name  = None
        self._phone = None
        self._email = None
        self.name   = name
        self.phone  = phone
        self.email  = email

    @property
    def name(self):  return self._name
    @name.setter
    def name(self, v):
        if not v.strip(): raise ValueError("Name empty")
        self._name = v.strip().title()

    @property
    def phone(self):  return self._phone
    @phone.setter
    def phone(self, v):
        clean = "".join(c for c in v if c.isdigit())
        if not 7 <= len(clean) <= 15: raise ValueError("Invalid phone")
        self._phone = clean

    @property
    def email(self):  return self._email
    @email.setter
    def email(self, v):
        if "@" not in v or "." not in v.split("@")[-1]:
            raise ValueError("Invalid email")
        self._email = v.lower().strip()

    def __str__(self):
        return f"{self._name} | {self._phone} | {self._email}"

## Challenge 2 — Vending Machine

```python
class VendingMachine:
    VALID_COINS = [10, 20, 50, 100, 200]     # in cent

    def __init__(self):
        self._prices = {}         # item name -> price in cent
        self._quantities = {}     # item name -> number left
        self._credit = 0          # cent inserted so far

    @property
    def credit(self):
        return self._credit       # read-only

    def add_item(self, name, price, quantity):
        if not name.strip():
            raise ValueError("Item name cannot be empty")
        if price <= 0:
            raise ValueError("Price must be positive")
        if quantity < 0:
            raise ValueError("Quantity cannot be negative")
        self._prices[name] = price
        self._quantities[name] = quantity

    def insert_coin(self, coin):
        if coin not in VendingMachine.VALID_COINS:
            raise ValueError(f"Coin not accepted: {coin}c")
        self._credit += coin

    def buy(self, name):
        if name not in self._prices:
            raise ValueError(f"No such item: {name}")
        if self._quantities[name] == 0:
            raise ValueError(f"{name} is sold out")
        price = self._prices[name]
        if self._credit < price:
            raise ValueError(f"Insert {price - self._credit}c more")
        self._quantities[name] -= 1
        change = self._credit - price
        self._credit = 0
        return change

    def cancel(self):
        refund = self._credit
        self._credit = 0
        return refund

    def __str__(self):
        lines = [f"Credit: €{self._credit / 100:.2f}"]
        for name in self._prices:
            lines.append(f"  {name:<10} €{self._prices[name] / 100:.2f}  ({self._quantities[name]} left)")
        return "\n".join(lines)
```

## Challenge 3 — Car Park

```python
import math

class CarPark:
    def __init__(self, name, capacity, hourly_rate):
        if not isinstance(capacity, int) or capacity <= 0:
            raise ValueError("Capacity must be a positive whole number")
        self._name = name
        self._capacity = capacity
        self._hourly_rate = 0
        self.hourly_rate = hourly_rate      # use the setter
        self._parked = []
        self._takings = 0

    @property
    def capacity(self):
        return self._capacity

    @property
    def hourly_rate(self):
        return self._hourly_rate

    @hourly_rate.setter
    def hourly_rate(self, value):
        if value <= 0:
            raise ValueError("Hourly rate must be positive")
        self._hourly_rate = value

    @property
    def spaces_free(self):
        return self._capacity - len(self._parked)

    @property
    def is_full(self):
        return self.spaces_free == 0

    @property
    def takings(self):
        return self._takings

    def enter(self, reg):
        reg = reg.strip().upper()
        if not reg:
            raise ValueError("Registration cannot be empty")
        if self.is_full:
            raise ValueError(f"{self._name} is full")
        if reg in self._parked:
            raise ValueError(f"{reg} is already parked")
        self._parked.append(reg)

    def leave(self, reg, hours):
        reg = reg.strip().upper()
        if reg not in self._parked:
            raise ValueError(f"{reg} is not parked here")
        if hours <= 0:
            raise ValueError("Hours must be positive")
        fee = math.ceil(hours) * self._hourly_rate
        self._parked.remove(reg)
        self._takings += fee
        return fee

    def parked_cars(self):
        return list(self._parked)     # a copy, not the real list

    def __str__(self):
        used = self._capacity - self.spaces_free
        return f"{self._name}: {used}/{self._capacity} spaces used, takings €{self._takings:.2f}"
```

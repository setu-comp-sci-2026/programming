# 7. Load knowledge into objects
Reuse your existing knowledge.json, or use this example (same schema):
~~~json
{
  "questions": [
    {
      "key": "power_on",
      "text": "Does the computer turn on?"
    },
    {
      "key": "display_on",
      "text": "Is there an image on the screen?"
    }
  ],
  "rules": [
    {
      "when": {
        "power_on": "no"
      },
      "advice": "Check the power cable and socket."
    },
    {
      "when": {
        "power_on": "yes",
        "display_on": "no"
      },
      "advice": "Check the monitor connection."
    },
    {
      "when": {
        "power_on": "yes",
        "display_on": "yes"
      },
      "advice": "Try restarting the computer or checking other symptoms."
    }
  ]
}
~~~
Create storage.py.

~~~python
import json
from pathlib import Path
from datetime import datetime
from question import Question
from rule import Rule
from troubleshooter import Troubleshooter

BASE_FOLDER = Path(__file__).resolve().parent
KNOWLEDGE_FILE = BASE_FOLDER / "knowledge.json"
HISTORY_FILE = BASE_FOLDER / "data" / "history.json"

def load_system():
    with open(KNOWLEDGE_FILE, "r", encoding="utf-8") as file:
        knowledge = json.load(file)
    system = Troubleshooter()
    for item in knowledge["questions"]:
        system.questions.append(Question(item["key"], item["text"]))
    for item in knowledge["rules"]:
        system.rules.append(Rule(item["when"], item["advice"]))
    return system
~~~

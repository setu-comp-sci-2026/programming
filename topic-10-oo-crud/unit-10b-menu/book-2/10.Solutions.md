### bank_account.py

~~~python
class BankAccount:
    def __init__(self, account_no, name, initial_balance=0):
        self._account_no = account_no
        self.name = name
        self._balance = 0
        self.balance = initial_balance

    @property
    def account_no(self):
        return self._account_no

    @property
    def name(self):
        return self._name

    @name.setter
    def name(self, value):
        if not value.strip():
            raise ValueError("Name cannot be blank")
        self._name = value.strip()

    @property
    def balance(self):
        return self._balance

    @balance.setter
    def balance(self, value):
        if value < 0:
            raise ValueError("Balance cannot be negative")
        self._balance = value

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("Deposit must be positive")
        self.balance = self.balance + amount

    def withdraw(self, amount):
        if amount <= 0 or amount > self.balance:
            raise ValueError("Invalid withdrawal")
        self.balance = self.balance - amount

    def __str__(self):
        return f"{self.account_no}: {self.name} - €{self.balance:.2f}"

    def to_dict(self):
        return {"account_no": self.account_no,
                "name": self.name, "balance": self.balance}
~~~

### bank.py

~~~python
from bank_account import BankAccount

class Bank:
    def __init__(self):
        self.accounts = []

    def list_accounts(self):
        for account in self.accounts:
            print(account)

    def find_account(self, account_no):
        for account in self.accounts:
            if account.account_no == account_no:
                return account
        return None

    def add_account(self, account):
        if self.find_account(account.account_no) is not None:
            return False
        self.accounts.append(account)
        return True

    def rename_account(self, account_no, new_name):
        account = self.find_account(account_no)
        if account is None or not new_name.strip():
            return False
        account.name = new_name.strip()  # property setter validates
        return True

    def delete_account(self, account_no):
        account = self.find_account(account_no)
        if account is None:
            return False
        self.accounts.remove(account)
        return True

    def count_accounts(self):
        return len(self.accounts)
~~~

### storage.py

~~~python
import json
from pathlib import Path
from datetime import datetime
from bank import Bank
from bank_account import BankAccount

DATA_FILE = Path(__file__).resolve().parent / "data" / "accounts.json"
LOG_FILE = DATA_FILE.parent / "bank.log"

def save_bank(bank):
    records = []
    for account in bank.accounts:
        records.append(account.to_dict())
    DATA_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(DATA_FILE, "w", encoding="utf-8") as file:
        json.dump(records, file, indent=4)

def load_bank():
    bank = Bank()
    if not DATA_FILE.exists():
        return bank
    with open(DATA_FILE, "r", encoding="utf-8") as file:
        records = json.load(file)
    for record in records:
        account = BankAccount(record["account_no"], record["name"], record["balance"])
        if not bank.add_account(account):
            raise ValueError("Duplicate account in saved data")
    return bank

def log_event(message):
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    stamp = datetime.now().isoformat(timespec="seconds")
    with open(LOG_FILE, "a", encoding="utf-8") as file:
        file.write(f"{stamp} {message}\n")
~~~

### main.py

~~~python
from bank_account import BankAccount
from storage import load_bank, save_bank, log_event

def show_menu():
    print("\n=== BANK ===")
    print("1. List accounts")
    print("2. Find account")
    print("3. Add account")
    print("4. Rename account")
    print("5. Delete account")
    print("6. Save accounts")
    print("0. Exit")

def main():
    try:
        bank = load_bank()
    except (OSError, ValueError, KeyError, TypeError) as error:
        print("Cannot load accounts:", error)
        return
    while True:
        show_menu()
        choice = input("Choose: ").strip()
        try:
            if choice == "1":
                bank.list_accounts()
            elif choice == "2":
                account = bank.find_account(input("Account number: ").strip())
                print(account if account is not None else "Not found")
            elif choice == "3":
                number = input("Account number: ").strip()
                if not number:
                    raise ValueError("Account number cannot be blank")
                name = input("Name: ")
                balance = float(input("Balance: "))
                if bank.add_account(BankAccount(number, name, balance)):
                    print("Added")
                    log_event("INFO Account added")
                else:
                    print("Duplicate account")
                    log_event("WARNING Duplicate account rejected")
            elif choice == "4":
                success = bank.rename_account(input("Account number: ").strip(), input("New name: "))
                print("Renamed" if success else "Rename failed")
                if success:
                    log_event("INFO Account renamed")
            elif choice == "5":
                success = bank.delete_account(input("Account number: ").strip())
                print("Deleted" if success else "Not found")
                if success:
                    log_event("INFO Account deleted")
            elif choice in ("6", "0"):
                save_bank(bank)
                print("Accounts saved")
                log_event("INFO Accounts saved")
                if choice == "0":
                    break
            else:
                print("Invalid choice")
        except (ValueError, OSError) as error:
            print("Operation error:", error)

if __name__ == "__main__":
    main()
~~~

## PC Troubleshooter solution

### question.py

~~~python
class Question:
    def __init__(self, key, text):
        self.key = key
        self.text = text

    def __str__(self):
        return self.text
~~~

### rule.py

~~~python
class Rule:
    def __init__(self, conditions, advice):
        self.conditions = conditions
        self.advice = advice

    def matches(self, answers):
        for key, expected in self.conditions.items():
            if answers.get(key) != expected:
                return False
        return True
~~~

### troubleshooter.py

~~~python
class Troubleshooter:
    def __init__(self):
        self.questions = []
        self.rules = []

    def recommend(self, answers):
        for rule in self.rules:
            if rule.matches(answers):
                return rule.advice
        return "No matching rule. Ask for further help."
~~~

### storage.py

~~~python
import json
from pathlib import Path
from datetime import datetime
from question import Question
from rule import Rule
from troubleshooter import Troubleshooter

BASE_FOLDER = Path(__file__).resolve().parent
KNOWLEDGE_FILE = BASE_FOLDER / "knowledge.json"
HISTORY_FILE = BASE_FOLDER / "data" / "history.json"

def load_system():
    with open(KNOWLEDGE_FILE, "r", encoding="utf-8") as file:
        knowledge = json.load(file)
    system = Troubleshooter()
    for item in knowledge["questions"]:
        system.questions.append(Question(item["key"], item["text"]))
    for item in knowledge["rules"]:
        system.rules.append(Rule(item["when"], item["advice"]))
    return system

def load_history():
    try:
        with open(HISTORY_FILE, "r", encoding="utf-8") as file:
            return json.load(file)
    except FileNotFoundError:
        return []

def save_history(answers, advice):
    history = load_history()
    history.append({"time": datetime.now().isoformat(timespec="seconds"),
                    "answers": answers, "advice": advice})
    HISTORY_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(HISTORY_FILE, "w", encoding="utf-8") as file:
        json.dump(history, file, indent=4)
~~~

### main.py

~~~python
from storage import load_system, load_history, save_history

def ask_questions(questions):
    answers = {}
    for question in questions:
        while True:
            answer = input(question.text + " (yes/no): ").strip().lower()
            if answer in ("yes", "no"):
                answers[question.key] = answer
                break
            print("Please enter yes or no.")
    return answers

def run_consultation(system):
    answers = ask_questions(system.questions)
    advice = system.recommend(answers)
    print("Recommendation:", advice)
    save_history(answers, advice)
    print("Consultation saved.")

def view_history():
    history = load_history()
    if not history:
        print("No consultations yet.")
    for record in history:
        print(record["time"], record["answers"], record["advice"])

def show_menu():
    print("\n=== PC TROUBLESHOOTER ===")
    print("1. Start a consultation")
    print("2. View consultation history")
    print("3. List knowledge-base questions")
    print("0. Exit")

def main():
    try:
        system = load_system()
    except (OSError, ValueError, KeyError, TypeError) as error:
        print("Cannot load knowledge:", error)
        return
    while True:
        show_menu()
        choice = input("Choose: ").strip()
        try:
            if choice == "1":
                run_consultation(system)
            elif choice == "2":
                view_history()
            elif choice == "3":
                for question in system.questions:
                    print(question)
            elif choice == "0":
                break
            else:
                print("Invalid choice")
        except (OSError, ValueError, KeyError, TypeError) as error:
            print("Operation failed:", error)

if __name__ == "__main__":
    main()
~~~

## Check your understanding
Bank() belongs outside the loop because one collection must survive all menu choices. to_dict creates a separate dictionary. JSON holds data, not methods. Loading reconstructs objects. A missing history file means no previous consultations; corrupt JSON is an error. First matching rule wins; old history records preserve the original advice. Logs append events; JSON snapshots replace the whole list.

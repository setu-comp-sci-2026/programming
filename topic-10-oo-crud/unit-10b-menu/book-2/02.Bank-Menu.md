# 2. Give the user a menu
The business logic already exists. We only need to call it. Add this to main.py:

~~~python
from bank import Bank
from bank_account import BankAccount

def show_menu():
    print("\n=== BANK ===")
    print("1. List accounts")
    print("2. Find account")
    print("3. Add account")
    print("4. Rename account")
    print("5. Delete account")
    print("6. Save accounts")
    print("0. Exit")

def main():
    bank = Bank()
    while True:
        show_menu()
        choice = input("Choose: ").strip()
        if choice == "1":
            bank.list_accounts()
        elif choice == "0":
            break
        else:
            print("Invalid choice")

if __name__ == "__main__":
    main()
~~~

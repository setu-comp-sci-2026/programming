## Complete class implementations

Copy this into a fresh file called `bank_solution.py` if you want to test the complete solution.

~~~python
class BankAccount:
    def __init__(self, account_no, name, balance=0):
        if balance < 0:
            raise ValueError("Balance cannot be negative")
        self.account_no = account_no
        self.name = name
        self.balance = balance

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("Deposit must be positive")
        self.balance += amount

    def withdraw(self, amount):
        if amount <= 0:
            raise ValueError("Withdrawal must be positive")
        if amount > self.balance:
            raise ValueError("Insufficient funds")
        self.balance -= amount

    def __str__(self):
        return f"{self.account_no}: {self.name} - €{self.balance:.2f}"


class Bank:
    def __init__(self):
        self.accounts = []

    def list_accounts(self):
        for account in self.accounts:
            print(account)

    def find_account(self, account_no):
        for account in self.accounts:
            if account.account_no == account_no:
                return account
        return None

    def add_account(self, account):
        if self.find_account(account.account_no) is not None:
            return False
        self.accounts.append(account)
        return True

    def rename_account(self, account_no, new_name):
        account = self.find_account(account_no)
        if account is None or new_name.strip() == "":
            return False
        account.name = new_name.strip()
        return True

    def delete_account(self, account_no):
        account = self.find_account(account_no)
        if account is None:
            return False
        self.accounts.remove(account)
        return True

    def count_accounts(self):
        return len(self.accounts)


# Example test code (no menu needed for Monday)
bank = Bank()
alice = BankAccount("A001", "Alice", 500)
bob = BankAccount("A002", "Bob", 1200)
charlie = BankAccount("A003", "Charlie", 250)

print(bank.add_account(alice))    # True
print(bank.add_account(bob))      # True
print(bank.add_account(charlie))  # True
print(bank.add_account(alice))    # False

alice.deposit(100)
alice.withdraw(50)
print(alice)                      # A001: Alice - €550.00

found = bank.find_account("A002")
if found is not None:
    print(found.name)             # Bob

print(bank.find_account("A999")) # None
print(bank.rename_account("A002", "Robert"))  # True
print(bank.rename_account("A001", "   "))    # False
print(bank.delete_account("A003"))           # True
print(bank.delete_account("A003"))           # False
print(bank.count_accounts())                  # 2
bank.list_accounts()
~~~

## Mini-exercise answers

**1. Create Charlie:**

~~~python
charlie = BankAccount("A003", "Charlie", 250)
print(charlie)
~~~

**2. Deposit and withdraw:**

~~~python
bob.deposit(200)
charlie.withdraw(75)
print(bob)
print(charlie)
~~~

**3. Object reference:** After `bank.accounts.append(alice)`, `alice.deposit(25)` changes the same object referenced by `bank.accounts[0]`.

**4. Search for an absent account:**

~~~python
found = bank.find_account("A999")
if found is None:
    print("Account not found")
~~~

Accessing `.name` on `None` raises an `AttributeError`.

**5. Add a new account and reject a duplicate:**

~~~python
print(bank.add_account(BankAccount("A004", "Dana", 700)))  # True
print(bank.add_account(BankAccount("A004", "Different Dana", 50)))  # False
~~~

**6. Rename and deposit:**

~~~python
print(bank.rename_account("A003", "Charles"))
account = bank.find_account("A001")
if account is not None:
    account.deposit(50)
~~~

**7. Delete and verify:**

~~~python
print(bank.delete_account("A003"))
print(bank.find_account("A003") is None)  # True
~~~

## Challenge D — responsibilities

`BankAccount` stores and validates **one** account's information. `Bank` stores a list of accounts and performs collection-level CRUD. This keeps responsibilities clear and allows the account class to be reused.

**Important limitation:** Direct assignments to attributes are still possible in this introductory design. Python properties/setters offer more control; they are outside today's lab.

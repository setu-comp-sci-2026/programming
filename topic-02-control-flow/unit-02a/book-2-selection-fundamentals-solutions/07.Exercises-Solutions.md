# Exercises - Solutions

## Exercise 1 - Voting age

~~~python
age = int(input("Enter your age: "))

if age >= 18:
    print("You are old enough to vote")
else:
    print("You are not old enough to vote")
~~~

## Exercise 2 - Free shipping

~~~python
order_total = float(input("Order total: €"))

if order_total >= 50:
    print("Free shipping")
else:
    print("Shipping charge applies")
~~~

## Exercise 3 - Pass or fail

~~~python
mark = int(input("Enter mark: "))

if mark >= 40:
    print("Pass")
else:
    print("Fail")
~~~

## Exercise 4 - PIN check

~~~python
pin = input("Enter PIN: ")

if pin == "1234":
    print("Correct PIN")
else:
    print("Incorrect PIN")
~~~

## Exercise 5 - Divisible by 5

~~~python
number = int(input("Enter a whole number: "))

if number % 5 == 0:
    print("Divisible by 5")
else:
    print("Not divisible by 5")
~~~

## Exercise 6 - Ticket price

~~~python
age = int(input("Enter age: "))
~~~

# Exercises - Solutions

## Exercise 1 - Number classification

~~~python
number = float(input("Enter a number: "))

if number < 0:
    print("Negative")
elif number == 0:
    print("Zero")
else:
    print("Positive")
~~~

## Exercise 2 - Largest of three

~~~python
a = float(input("First number: "))
b = float(input("Second number: "))
c = float(input("Third number: "))

if a > b and a > c:
    print(a, "is largest")
elif b > a and b > c:
    print(b, "is largest")
else:
    print(c, "is largest")
~~~

## Exercise 3 - Delivery charge

~~~python
order_total = float(input("Order total: €"))

if order_total >= 50:
    delivery = 0
elif order_total >= 25:
    delivery = 4
else:
    delivery = 7

print("Delivery charge: €", delivery)
~~~

## Exercise 5 - Weekend discount

~~~python
day = input("Day: ").lower()
age = int(input("Age: "))

if (day == "saturday" or day == "sunday") and age < 18:
    print("Discount applies")
else:
    print("No discount")
~~~

## Exercise 6 - Valid percentage

~~~python
percentage = float(input("Percentage: "))

if percentage < 0 or percentage > 100:
    print("Invalid")
elif percentage >= 70:
    print("High")
elif percentage >= 40:
    print("Medium")
else:
    print("Low")
~~~

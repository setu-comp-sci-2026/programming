## Exercise 5

```python
import json

def load_students():
    try:
        with open("students.json", "r", encoding="utf-8") as file:
            return json.load(file)
    except FileNotFoundError:
        print("No student file yet")
        return []
    except json.JSONDecodeError:
        print("The student file is damaged. Do not overwrite it.")
        return None
    except OSError as error:
        print("Could not read student file:", error)
        return None

students = load_students()
if students is None:
    print("Loading failed. Please check the file.")
else:
    print(students)
```

## Challenges 2 and 3

```python
import json

def load_products():
    try:
        with open("products.json", "r", encoding="utf-8") as file:
            return json.load(file)
    except FileNotFoundError:
        return []
    except json.JSONDecodeError:
        print("Invalid JSON: existing data must be repaired")
        return None
    except OSError as error:
        print("Cannot load products:", error)
        return None

def save_products(products):
    try:
        with open("products.json", "w", encoding="utf-8") as file:
            json.dump(products, file, indent=4)
    except OSError as error:
        print("Cannot save products:", error)
        return False
    return True

products = load_products()
if products is None:
    print("Stopping to protect existing data")
else:
    products.append({"code": 101, "name": "Mouse", "price": 12.50})
    if save_products(products):
        print("Saved successfully")
```

**Why?** If a damaged file is treated as an empty list and saved, the original data may be erased.

## Challenge 5 — One possible solution

```python
import json

def get_menu_choice():
    valid = False
    while not valid:
        try:
            choice = int(input("1 View | 2 Add | 3 Save | 4 Exit: "))
            if 1 <= choice <= 4:
                valid = True
            else:
                print("Choose 1–4")
        except ValueError:
            print("Enter a whole number")
    return choice

def save_dogs(dogs):
    try:
        with open("dogs.json", "w", encoding="utf-8") as file:
            json.dump(dogs, file, indent=4)
        return True
    except OSError as error:
        print("Save failed:", error)
        return False

dogs = []
finished = False
while not finished:
    choice = get_menu_choice()
    if choice == 1:
        print(dogs)
    elif choice == 2:
        name = input("Dog name: ")
        dogs.append({"name": name})
    elif choice == 3:
        if save_dogs(dogs):
            print("Saved!")
    else:
        finished = True
```

**Extension:** Load saved records safely at startup, and do not allow saving if loading failed because a file was damaged.
